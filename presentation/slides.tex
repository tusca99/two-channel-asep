\section{Motivation}
\begin{frame}{Why study exclusion processes?}
  \begin{itemize}
    \item Asymmetric Simple Exclusion Process (TASEP) is the reference model
      of \emph{non-equilibrium} transport --- the role of the Ising model for
      equilibrium statistical mechanics.
  \end{itemize}
\centering
\includegraphics[width=0.8\textwidth]{Kynesin_dynein.png}
  \begin{block}{Motivation for two channels}
    In cells, kinesin and dynein walk \emph{in opposite directions} on the
    same microtubule. Two lanes of particles moving in opposite directions,
    coupled at the boundaries --- can they \emph{break symmetry}
    spontaneously?
  \end{block}
\end{frame}

    \textbf{Symmetric:} $\rho_1 = \rho_2$ (both lanes equal).\\

    the other dilute --- $\rho_1 \neq \rho_2$.
  \end{block}
  \note{Two identical lanes, identical dynamics, identical injection. The
    natural guess is a symmetric steady state. The surprising result is that
    the steady state can instead be asymmetric --- one lane dense, the other
    dilute --- even though the rules are exactly symmetric: spontaneous
    symmetry breaking. The dashed lines are the key ingredient --- the narrow
    entrance: you can only enter a lane if the exit site of the OTHER lane is
    empty. The two lanes share a bottleneck at each end.}
\end{frame}

\section{The Model}
\begin{frame}{The model: two-channel ASEP with narrow entrances}
  \begin{center}
    \href{lattice_animation.mp4}{\includegraphics[width=0.82\textwidth]{poster_lattice.png}}
  \end{center}
  \vspace{0.1em}
  \centering
  \small \emph{Click the frame to play the animation}
  (\texttt{lattice\_animation.mp4}).
  \note{The model in action. Channel 1 (top) enters at the left and hops
    right with rate 1; channel 2 (bottom) enters at the right and hops left
    with rate 1. Hard-core exclusion: at most one particle per site. One
    frame is one event (Gillespie). Watch the waiting dots at the ends: an
    entrance is blocked while the OTHER lane's exit site is occupied --- that
    is the narrow-entrance coupling.}
\end{frame}

\begin{frame}{The narrow-entrance rule}
  \begin{itemize}
    \item Enter channel 1 (rate $\alpha$) iff site 1 of channel 1 is empty
      \textbf{and} the exit site of channel 2 is empty.
    \item Enter channel 2 (rate $\alpha$) iff site L of channel 2 is empty
      \textbf{and} the exit site of channel 1 is empty.
  \end{itemize}
  \vspace{0.4em}
  \begin{center}

    \end{tikzpicture}
  \end{center}
  \vspace{0.2em}
  \begin{block}{The coupling}
    Each channel's \emph{entrance} is coupled to the other channel's
    \emph{exit} --- a shared bottleneck at each end.
  \end{block}
  \note{The only non-trivial rule. Bulk hops are rate 1 to the next site if
    empty; exits are rate beta, independent of the other channel. The
    coupling is entirely at the entrances: to enter a lane you need the OTHER
    lane's exit site to be free. Everything interesting in this model comes
    from this one rule.}
\end{frame}

\begin{frame}{What we want to compute}
  Steady-state properties as a function of the rates $(\alpha, \beta)$:
  \begin{itemize}
    \item $\alpha$ = entrance rate of either channel (identical for both).
    \item $\beta$ = exit rate of either channel (identical for both).
    \item Phase diagram in the $(\alpha, \beta)$ plane.
    \item Bulk densities $\rho_1, \rho_2$ and currents $J_1, J_2$.
    \item The joint density distribution $P(\rho_1, \rho_2)$ --- the diagnostic
      for spontaneous symmetry breaking (SSB).
  \end{itemize}
  \note{Approach: mean-field theory solves the model analytically; Monte
    Carlo checks the theory. P(rho1,rho2) is the SSB diagnostic: a single
    peak means the two lanes are equal, two peaks means they have broken
    apart.}
\end{frame}

\section{TASEP (recap)}
\begin{frame}{Recap: single-lane TASEP}
  One channel, particles enter with rate $\alpha$, hop right with rate 1,
  exit with rate $\beta$.
  \vspace{0.3em}
  \begin{block}{Mean-field approximation}
    Assume no correlations between sites (eq 1 of the paper):
    \[
      \langle p_i p_j \rangle = \langle p_i \rangle\langle p_j \rangle
      = p_i p_j .
    \]
  \end{block}
  The current across bond $i \to i{+}1$ is then
  \[
    J = \langle p_i (1 - p_{i+1})\rangle = \rho(1-\rho), \qquad \rho = \langle p_i\rangle .
  \]
  \note{Three stationary phases depending on which boundary is rate-limiting:
    LD when the entrance limits, HD when the exit limits, MC when the bulk
    limits. This is the single-lane TASEP we build on.}
\end{frame}

\section{MFT for the two-channel ASEP}
\begin{frame}{Strategy: two single-lane TASEPs with effective rates}
  \footnotesize
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}
  Two single-lane TASEPs coupled \emph{only through the entrances} (eq 2);
  the \emph{exit} rate $\beta$ is identical for both, only the entrance rates
  differ.
  \begin{block}{Effective entrance rates}
    \footnotesize
    \begin{align*}
      \alpha_1 &= \alpha\,\langle 1 - m_1\rangle = \alpha (1 - m_1), \\
      \alpha_2 &= \alpha\,\langle 1 - p_L\rangle = \alpha (1 - p_L),
    \end{align*}
    $\alpha_1,\alpha_2$ = entrance rates felt by channels 1, 2; $m_1$ =
    channel 2's exit density, $p_L$ = channel 1's exit density.
  \end{block}
  Each channel is a standard TASEP with rates $(\alpha_1,\beta)$,
  $(\alpha_2,\beta)$ --- whose steady states we already know.
  \begin{block}{Nine candidate phases}
    \scriptsize
    Each channel in LD/HD/MC $\Rightarrow$ $3\times3$; filter by
    self-consistency of $\alpha_1,\alpha_2$:
    \[
      \setlength{\arraycolsep}{4pt}
      \begin{array}{c|ccc}
        \text{ch.2} \backslash \text{ch.1} & \mathrm{LD} & \mathrm{HD} & \mathrm{MC}\\
        \hline
        \mathrm{LD} & \mathbf{LD/LD} & \mathbf{HD/LD} & \mathrm{MC/LD}\\
        \mathrm{HD} & \mathbf{LD/HD} & \mathrm{HD/HD} & \mathrm{MC/HD}\\
        \mathrm{MC} & \mathrm{LD/MC} & \mathrm{HD/MC} & \mathrm{MC/MC}
      \end{array}
    \]
    Only the \textbf{bold} combinations survive.
  \end{block}
  \note{Each channel is a standard single-lane TASEP with its own effective
    entrance rate, sitting in LD, HD or MC: nine candidate combinations. The
    mean-field calculation checks each of the nine for self-consistency. Only
    LD/LD and HD/LD survive as broken phases.}
\end{frame}

\begin{frame}{Symmetric phases: set-up}
  \small
  Notation: $p_i$ = density at site $i$ of channel 1, $m_i$ = density at site
  $i$ of channel 2, both counted from their own entrance ($i=1,\dots,L$).
  So $m_1$ is channel 2's exit density, $p_L$ is channel 1's exit density.
  \begin{block}{Symmetric ansatz (eq 6-7)}
    \[
      J_1 = J_2, \quad \rho_1 = \rho_2, \quad \alpha_1 = \alpha_2, \qquad
      p_i = m_{L-i+1}.
    \]
  \end{block}
  \begin{itemize}
    \item Both channels in the same single-lane phase.
    \item Symmetry + the exit current relation determine the effective rates
      \emph{self-consistently}.
    \item Key relation (exit current of channel 2, eq 8/11):
      \[
        J_2 = \beta\, m_1 .
      \]
  \end{itemize}
  \note{The exit current of channel 2 is beta times m_1, its exit-site
    density. The J_2 = beta m_1 relation is the workhorse of the whole
    calculation.}
\end{frame}

\begin{frame}{Symmetric MC phase (eqs 8--10)}
  \footnotesize
  \textbf{Step 1: current fixes $m_1$ (eq 8).}
  In MC, $J_1 = J_2 = 1/4$; the exit current of channel 2 gives
  \[
    J_2 = \beta m_1 = \frac14 \quad\Rightarrow\quad m_1 = \frac{1}{4\beta}.
  \]
  (Physical range: $\beta > 0$.)\\
  \textbf{Step 2: effective entrance rate (eq 9).}
  \[
    \alpha_1 = \alpha(1 - m_1) = \alpha\left(1 - \frac{1}{4\beta}\right).
  \]
  \textbf{Step 3: existence condition (eq 10).}
  MC needs $\alpha_1 > 1/2$ and $\beta > 1/2$:
  \[
    \alpha\left(1 - \frac{1}{4\beta}\right) > \frac12
    \quad\Longrightarrow\quad
    \alpha > \frac{2\beta}{4\beta - 1}
  \]
\end{frame}

\begin{frame}{Symmetric LD phase (eqs 11--13)}
  \footnotesize
  \textbf{Step 1: current fixes $m_1$ (eq 11).}
  In LD, $J_1 = J_2 = \alpha_1(1-\alpha_1)$; exit current of channel 2:
  \[
    J_2 = \beta m_1 = \alpha_1(1-\alpha_1)
    \quad\Rightarrow\quad
    m_1 = \frac{\alpha_1(1-\alpha_1)}{\beta}.
  \]
  \textbf{Step 2: self-consistency for $\alpha_1$ (eq 2).}
  \[
    \alpha_1 = \alpha(1 - m_1)
    = \alpha\left(1 - \frac{\alpha_1(1-\alpha_1)}{\beta}\right).
  \]
  A quadratic in $\alpha_1$; physical root (eq 12):
  \[
    \alpha_1 = \frac{\alpha + \beta - \sqrt{(\alpha+\beta)^2 - 4\alpha^2\beta}}{2\alpha}.
  \]
  \textbf{Step 3: existence condition (eq 13).}
  LD needs $\alpha_1 < 1/2$ and $\alpha_1 < \beta$:
  \[
    \alpha < \frac{2\beta}{4\beta - 1}
  \]
\end{frame}

\begin{frame}{Symmetric HD phase: does not exist}
  \begin{itemize}
    \item In HD, $J_1 = J_2 = \beta(1-\beta)$. From $J_2 = \beta m_1$:
      \[
        m_1 = 1 - \beta.
      \]
    \item Effective entrance rate:
      \[
        \alpha_1 = \alpha(1 - m_1) = \alpha\beta.
      \]
    \item HD requires $\alpha_1 > \beta$, i.e.\ $\alpha\beta > \beta$, i.e.\
      $\alpha > 1$. But $\alpha \le 1$ by definition.
  \end{itemize}
  \begin{block}{Conclusion}
    \textbf{Symmetric HD phase is impossible} in the two-channel model with
    narrow entrances. Only MC and LD survive as symmetric phases.
  \end{block}
\end{frame}

\begin{frame}{Asymmetric phases: MC/HD and MC/LD don't exist}
  \small
  \begin{itemize}
    \item \textbf{MC/HD} (eq 15): requires simultaneously $\beta>1/2$
      (for MC) and $\beta<1/2$ (for HD) --- contradictory.
    \item \textbf{MC/LD} (eqs 16--19): MC needs $\alpha_1 > 1/2$, LD needs
      $\alpha_2 < 1/2$. Using $J_1 = \beta p_L = 1/4$ and
      $J_2 = \beta m_1 = \alpha_2(1-\alpha_2)$ leads to a condition (eq 19)
      that cannot hold for $0 < \alpha \le 1$, $0 < \beta \le 1$.
  \end{itemize}
  \begin{block}{Conclusion}
    An asymmetric phase must pair \textbf{one channel in LD} with the other in
    a different phase: the candidate is HD/LD.
  \end{block}
\end{frame}

\begin{frame}{Asymmetric HD/LD phase (eqs 20--23)}
  \small
  \textbf{Conditions (eq 20).}
  Channel 1 in HD: $\alpha_1 > \beta$, $\beta < 1/2$;\quad
  Channel 2 in LD: $\alpha_2 < \beta$, $\alpha_2 < 1/2$.
  \\
\textbf{Step 1: currents (eq 21).}
  \[
    J_1 = \beta p_L = \beta(1-\beta)
    \quad\Rightarrow\quad p_L = 1 - \beta,
    \qquad
    J_2 = \beta m_1 = \alpha_2(1-\alpha_2).
  \]
  \textbf{Step 2: effective rates (eq 22).}
  \[
    \alpha_2 = \alpha(1 - p_L) = \alpha\beta, \qquad
    \alpha_1 = \alpha(1 - m_1) = \alpha(1 - \alpha + \alpha^2\beta).
  \]
  \textbf{Step 3: existence (eq 23).}
  HD condition $\alpha_1 > \beta$:
  \[
    \alpha(1 - \alpha + \alpha^2\beta) > \beta
    \quad\Longrightarrow\quad
    \beta < \frac{\alpha}{1 + \alpha + \alpha^2}
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: set-up (eqs 24--26)}
  \small
  Both channels in LD, but with \emph{different} densities and currents.
  \begin{block}{Conditions (eq 24)}
    $\alpha_1 < \beta$, $\alpha_1 < 1/2$ and $\alpha_2 < \beta$, $\alpha_2 < 1/2$.
  \end{block}
  \begin{block}{Currents and effective rates (eqs 25--26)}
    \[
      J_1 = \beta p_L = \alpha_1(1-\alpha_1), \qquad
      J_2 = \beta m_1 = \alpha_2(1-\alpha_2),
    \]
    \[
      p_L = \frac{\alpha_1(1-\alpha_1)}{\beta}, \qquad
      m_1 = \frac{\alpha_2(1-\alpha_2)}{\beta},
    \]
    \[
      \alpha_1 = \alpha\left(1 - \frac{\alpha_2(1-\alpha_2)}{\beta}\right), \qquad
      \alpha_2 = \alpha\left(1 - \frac{\alpha_1(1-\alpha_1)}{\beta}\right).
    \]
  \end{block}
  Two coupled quadratics in $\alpha_1, \alpha_2$. Simplify with symmetric and
  antisymmetric combinations (eq 27):
  \[
    S = \alpha_1 + \alpha_2, \qquad D = \alpha_1 - \alpha_2 \neq 0.
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: solve (eqs 27--31)}
  \footnotesize
  \textbf{Step 1: difference of the two equations (eq 28).}
  \[
    D = \frac{\alpha}{\beta}\left(\alpha_1 - \alpha_1^2 - \alpha_2 + \alpha_2^2\right)
    = \frac{\alpha}{\beta} D (1 - S).
  \]
  Since $D \neq 0$ (asymmetric phase), we get
  \[
    S = 1 - \frac{\beta}{\alpha} \qquad \text{(eq 29)}
  \]
  \textbf{Step 2: sum of the two equations (eq 30).}
  \[
    S = 2\alpha - \frac{\alpha}{\beta}\left[S - \frac{S^2 + D^2}{2}\right]
    \qquad\text{(using $\alpha_1^2 + \alpha_2^2 = (S^2+D^2)/2$).}
  \]
  \textbf{Step 3: substitute $S = 1 - \beta/\alpha$ (eq 31).}
  \[
    D^2 = 1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: effective rates (eq 32)}
  \scriptsize
  From $S$ and $D$ we reconstruct the two effective entrance rates:
  \[
    \alpha_1 = \frac{S + D}{2}, \qquad \alpha_2 = \frac{S - D}{2}.
  \]
  \begin{align*}
    \alpha_1 &= \frac{1}{2}\left[1 - \frac{\beta}{\alpha} + \sqrt{1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2}\right], \\[0.4em]
    \alpha_2 &= \frac{1}{2}\left[1 - \frac{\beta}{\alpha} - \sqrt{1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2}\right].
  \end{align*}
  \begin{block}{Existence (eq 33)}
    Real roots need $D^2 > 0$; together with the LD conditions (eq 24):
    \[
      \frac{\alpha}{1 + \alpha + \alpha^2} < \beta <
        \frac{\alpha(1-2\alpha) + 2\alpha\sqrt{\alpha^2 - \alpha + 1}}{3}
    \]
  \end{block}
\end{frame}

\begin{frame}{Why does mean-field fail here?}
  \begin{block}{The source of the discrepancy}
    Mean-field assumes \emph{no correlations} between sites (eq 1). The
    narrow-entrance coupling ties each entrance to the other channel's exit
    site --- a strongly-correlated, local ``impurity'' at the boundary that
    mean-field cannot capture.
  \end{block}
  \begin{itemize}
    \item These boundary correlations \emph{propagate into the bulk} and
      shift the effective phase boundaries.
    \item The paper's own conclusion (Sec.~4): symmetry breaking in this
      model is a \emph{boundary-induced} phenomenon.
    \item The deviation \textbf{persists up to $L=2000$} at
      $5\times10^{4}$ steps/site --- looks like a genuine MFT failure,
      not a transient; the thermodynamic limit is still open.
  \end{itemize}
\end{frame}

\section{Computational methods}
\begin{frame}{Simulations I: why BKL, why Fenwick}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \textbf{BKL rate-proportional (rejection-free)}
      \begin{itemize}
        \item Pick next event $\propto$ rate — no rejections, exact Gillespie \cite{bkl1975}.
        \item Classic: scan active sites $\propto n_{\mathrm{active}}$ — cheap at low $\beta$, \con{costly at high density}.
        \item \pos{Fenwick tree} \cite{fenwick1994}: binary indexed tree, $O(\log L)$ pick/update — \textbf{${\sim}2.4\times$ faster, identical physics}.
      \end{itemize}
      \vspace{0.3em}
      {\small Cost crosses in $\beta$: BKL wins dilute, Fenwick wins dense.}
    \column{0.45\textwidth}
      \centering
      \begin{tabular}{lc}
        \toprule
        Kernel & Msteps/s (1 core) \\
        \midrule
        classic BKL & 1.8 \\
        Fenwick BKL & 4.4 \\
        \bottomrule
      \end{tabular}
      \vspace{0.6em}

      {\small \textbf{Why costly:} equilibration $\propto cL^2$ steps — $10^8$ at $L=2000$, $c=50$.}
  \end{columns}
  \note{BKL is exact, not Metropolis. Fenwick is the log-L trick that makes the high-density phase affordable. Crossing in beta is why we benchmarked both.}
\end{frame}

\begin{frame}{Simulations II: the GPU ensemble}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{One thread = one replica}
      \[
        \text{GPU}:\; \mathrm{rep}_1 \oplus \cdots \oplus \mathrm{rep}_N
      \]
      \begin{itemize}
        \item Serial chain $\to$ GPU is \con{latency-bound} ($\sim35$k/s/thread).
        \item \pos{Ensemble} $\to$ GPU is perfect: 2048 replicas in parallel.
        \item Exactly what SSB needs: $P(\rho_1,\rho_2)$ is an ensemble distribution, not a time average.
      \end{itemize}
    \column{0.47\textwidth}
      \centering
      \begin{tabular}{lc}
        \toprule
        Backend & Msteps/s agg. \\
        \midrule
        CPU 1 thread (Fenwick) & 4.4 \\
        CPU 30 workers & ${\sim}100$ \\
        GPU ensemble (RTX 4070) & 120--180 \\
        \bottomrule
      \end{tabular}
      \vspace{0.4em}
      {\scriptsize Per-$L$ bundles \texttt{results/<L>/} {\small +} \texttt{params.json}; plots read \texttt{.npz} — no SIM at plot time, reproducible, crash-safe.}
  \end{columns}
  \note{Long single trajectories: CPU still $\sim100\times$ faster per-thread; GPU wins only via mass parallelism. Lecture cue: this is why we have 512–2048 replicas for Fig3 — impossible on CPU. 40 beta scan at 200M steps/pt is a week on 1 core, hours on ensemble.}

      \textbf{\pos{Coupling is the engine}}\\{\small Zhu \textit{et al.} PRE \textbf{85}, 041132 (2012)}
      \begin{itemize}
        \item Leaky entrance: inject at $p\alpha$ also if other lane's exit is occupied.
        \item $p{=}0$ our model, $p{=}1$ decoupled; SSB grows with coupling.
        \item \pos{Both broken phases die at $p_c\approx0.6$} before $p{=}1$.
        \item {\small Flipping time $\tau\propto e^{L}$: SSB genuine.}
      \end{itemize}
    \column{0.33\textwidth}
      \textbf{\HL{Is SSB real at $p{=}0$?}}\\{\small Tian \textit{et al.} CPB \textbf{26}, 020503 (2017)}
      \begin{itemize}
        \item $N$-cluster MF + current minimization: \con{LD/LD should not exist} ($J$ highest in band).
        \item Boundary vs $N$ decays $\sim e^{aN}$: $\beta_c(\infty)\!=\!0.289$ at $\alpha\!=\!0.9$ (MF 0.332, SIM $\approx$0.25).
        \item {\small Yet their $L{=}10^4$ MC still sees LD/LD --- verdict open.}
      \end{itemize}
    \column{0.33\textwidth}
      \textbf{Our next}
      \begin{itemize}
        \item 2-site $\to$ 4-site cluster (Tian: N$\ge$5 adds only spurious sol.~II).
        \item SIM to $L=4000$--$20000$ at $c\approx100$ ($cL^2$); track $\mathrm{std}(\rho_1{-}\rho_2)$, $\tau(L)$.

  \begin{block}{Take-away}
    \small\textbf{Entrance coupling drives SSB; scaling + current minimization decide what survives.}
  \end{block}

\begin{frame}{Backup: equilibration must scale with $L$}
  \begin{itemize}
    \item A fixed steps/site budget \emph{under-equilibrates} large $L$:
      with $3\times10^{3}$/site, $L=500$ gives dense ${\sim}0.75$ vs
      $L=200$'s ${\sim}0.92$ --- a systematic bias, not noise.
    \item Calibration probe: the SSB dense basin at $L=1000$ needs
      ${\sim}10^{5}$ steps/site to saturate (dense ${\approx}0.89$).
    \item Rule: steps/site $\approx c\,L$ with $c\approx100$
      $\Rightarrow$ total $\approx c\,L^2$.
    \item Today's runs use $c=50$ ($5\times10^{4}$/site); the fig-5 drift
      at $L\ge4000$ is exactly this budget running out.
  \end{itemize}
