\section{Motivation}
\begin{frame}{Why study exclusion processes?}
  \begin{itemize}
    \item Asymmetric Simple Exclusion Process (TASEP) is the reference model
      of \emph{non-equilibrium} transport --- the role of the Ising model for
      equilibrium statistical mechanics.
    \item Minimal description of interacting particles on a 1D lattice with
      hard-core exclusion.
    \item Applications: motor proteins on cytoskeletal filaments
      (kinesin/dynein), ribosome motion in mRNA translation, car traffic,
      nanopore transport.
  \end{itemize}
  \vspace{0.5em}
  \begin{block}{Motivation for two channels}
    In cells, kinesin and dynein walk \emph{in opposite directions} on the
    same microtubule. Two lanes of particles moving in opposite directions,
    coupled at the boundaries --- can they \emph{break symmetry}
    spontaneously?
  \end{block}
\end{frame}

    \textbf{Symmetric:} $\rho_1 = \rho_2$ (both lanes equal).\\

  \end{block}

    natural guess is a symmetric steady state: both lanes at the same density.
    The surprising result is that the steady state can instead be asymmetric
    --- one lane dense, the other dilute --- even though the rules are exactly
    symmetric. This is spontaneous symmetry breaking, the two-lane variant of
    the Bridge model from class (broken in both LD/LD and LD/HD). The diagram
    shows the setup: channel 1 enters left and hops right, channel 2 enters
    right and hops left. The dashed lines are the key ingredient --- the narrow
    entrance: you can only enter a lane if the exit site of the OTHER lane is
    empty. So the two lanes share a bottleneck at each end.}
\end{frame}

\section{The Model}
\begin{frame}{The model: two-channel ASEP with narrow entrances}
  \begin{center}
    \movie[autostart,loop,width=0.82\textwidth]{\includegraphics[width=0.82\textwidth]{poster_lattice.png}}{lattice_animation.mp4}
  \end{center}
  \note{This is the model in action. Channel 1 (top) enters at the left and
    hops right with rate 1; channel 2 (bottom) enters at the right and hops
    left with rate 1. Hard-core exclusion: at most one particle per site.
    One frame is one event (Gillespie). Watch the waiting dots at the ends:
    an entrance is blocked while the OTHER lane's exit site is occupied ---
    that is the narrow-entrance coupling.}
\end{frame}

\begin{frame}{The narrow-entrance rule}
  \begin{itemize}
    \item Enter channel 1 (rate $\alpha$) iff site 1 of channel 1 is empty
      \textbf{and} the exit site of channel 2 is empty.
    \item Enter channel 2 (rate $\alpha$) iff site L of channel 2 is empty
      \textbf{and} the exit site of channel 1 is empty.
  \end{itemize}
  \vspace{0.4em}
  \begin{center}

    \end{tikzpicture}
  \end{center}
  \vspace{0.3em}
  \begin{block}{The coupling}
    Each channel's \emph{entrance} is coupled to the other channel's
    \emph{exit} --- a shared bottleneck at each end.
  \end{block}
  \note{The only non-trivial rule. Bulk hops are rate 1 to the next site if
    empty; exits are rate beta, independent of the other channel. The coupling
    is entirely at the entrances: to enter a lane you need the OTHER lane's
    exit site to be free. This is the narrow entrance --- a bottleneck shared
    by both lanes at each end. Everything interesting in this model comes from
    this one rule.}
\end{frame}

\begin{frame}{What we want to compute}
  Steady-state properties as a function of the rates $(\alpha, \beta)$:
  \begin{itemize}
    \item $\alpha$ = entrance rate of either channel (identical for both).
    \item $\beta$ = exit rate of either channel (identical for both).
    \item Phase diagram in the $(\alpha, \beta)$ plane.
    \item Bulk densities $\rho_1, \rho_2$ and currents $J_1, J_2$.
    \item The joint density distribution $P(\rho_1, \rho_2)$ --- the diagnostic
      for spontaneous symmetry breaking (SSB).
  \end{itemize}
  \note{Let me define the parameters up front. alpha is the entrance rate, beta
    the exit rate --- identical for the two channels. What we want out of the
    model is the phase diagram in the (alpha,beta) plane, the bulk densities
    rho1, rho2 and currents J1, J2 of each channel, and especially the joint
    density distribution P(rho1,rho2). That last one is the diagnostic for
    symmetry breaking: a single peak means the two lanes are equal, two peaks
    means they have broken apart. Approach: mean-field theory solves the model
    analytically; Monte Carlo checks the theory.}
\end{frame}

\section{Single-lane TASEP (recap)}
\begin{frame}{Recap: single-lane TASEP}
  One channel, particles enter with rate $\alpha$, hop right with rate 1,
  exit with rate $\beta$.
  \vspace{0.3em}
  \begin{block}{Mean-field approximation}
    Assume no correlations between sites (eq 1 of the paper):
    \[
      \langle p_i p_j \rangle = \langle p_i \rangle\langle p_j \rangle
      = p_i p_j .
    \]
  \end{block}
  The current across bond $i \to i{+}1$ is then
  \[
    J = \langle p_i (1 - p_{i+1})\rangle = \rho(1-\rho), \qquad \rho = \langle p_i\rangle .
  \]
  \note{Three stationary phases depending on which boundary is rate-limiting:
    LD when the entrance limits, HD when the exit limits, MC when the bulk
    limits. This is the single-lane TASEP we will build on.}
\end{frame}

\begin{frame}{Single-lane phases and phase diagram}
  \small
  \begin{columns}
    \column{0.5\textwidth}
      \centering
      \includegraphics[width=0.95\textwidth]{figures/tasep_phases.png}
    \column{0.5\textwidth}
      \footnotesize
      \begin{tabular}{lll}
        \toprule
        Phase & condition & $J$, $\rho$ \\
        \midrule
        LD & $\alpha<\beta$, $\alpha<1/2$ & $J{=}\alpha(1{-}\alpha)$, $\rho{=}\alpha$ \\
        HD & $\beta<\alpha$, $\beta<1/2$ & $J{=}\beta(1{-}\beta)$, $\rho{=}1{-}\beta$ \\
        MC & $\alpha>1/2$, $\beta>1/2$ & $J{=}1/4$, $\rho{=}1/2$ \\
        \bottomrule
      \end{tabular}

      \vspace{0.3em}
      \footnotesize
      LD = low density, HD = high density, MC = maximal current.

      \vspace{0.3em}
      For a single lane, \textbf{mean-field (MFT) is exact} (matrix ansatz).
  \end{columns}
\end{frame}

\begin{frame}{Single-lane currents}
  \centering
  \includegraphics[width=0.8\textwidth]{figures/tasep_current.png}
  \vspace{0.3em}
  \begin{itemize}
    \item At $\alpha > 1/2$, increasing $\beta$ drives LD $\to$ MC through the
      maximal-current value $J=1/4$.
    \item The current saturates: the bulk becomes the rate-limiting step.
    \item All curves collapse onto $J=1/4$ for $\alpha>1/2$ --- no overlap for

  \end{itemize}
  \note{Here J is the single-lane current in mean-field, MFT. When alpha is
    below 1/2 the current sits on the curve alpha(1-alpha); when alpha exceeds
    1/2 the current saturates at the maximal value 1/4. So curves with
    different alpha >= 1/2 collapse onto the same plateau --- that is expected,
    not an error. I will add an intermediate alpha and an inset zoom later once
    the theory section is finalized.}
\end{frame}

\section{Mean-field theory for the two-channel model}
\begin{frame}{Strategy: two single-lane TASEPs with effective rates}
  \footnotesize
  \setlength{\abovedisplayskip}{2pt}
  \setlength{\belowdisplayskip}{2pt}
  Two single-lane TASEPs coupled \emph{only through the entrances} (eq 2);
  the \emph{exit} rate $\beta$ is identical for both, only the entrance rates
  differ.
  \begin{block}{Effective entrance rates}
    \footnotesize
    \begin{align*}
      \alpha_1 &= \alpha\,\langle 1 - m_1\rangle = \alpha (1 - m_1), \\
      \alpha_2 &= \alpha\,\langle 1 - p_L\rangle = \alpha (1 - p_L),
    \end{align*}
    $\alpha_1,\alpha_2$ = entrance rates felt by channels 1, 2; $m_1$ =
    channel 2's exit density, $p_L$ = channel 1's exit density.
  \end{block}
  Each channel is a standard TASEP with rates $(\alpha_1,\beta)$,
  $(\alpha_2,\beta)$ --- whose steady states we already know. \pause
  \begin{block}{Nine candidate phases}
    \scriptsize
    Each channel in LD/HD/MC $\Rightarrow$ $3\times3$; filter by
    self-consistency of $\alpha_1,\alpha_2$:
    \[
      \setlength{\arraycolsep}{4pt}
      \begin{array}{c|ccc}
        \text{ch.2} \backslash \text{ch.1} & \mathrm{LD} & \mathrm{HD} & \mathrm{MC}\\
        \hline
        \mathrm{LD} & \mathbf{LD/LD} & \mathbf{HD/LD} & \mathrm{MC/LD}\\
        \mathrm{HD} & \mathbf{LD/HD} & \mathrm{HD/HD} & \mathrm{MC/HD}\\
        \mathrm{MC} & \mathrm{LD/MC} & \mathrm{HD/MC} & \mathrm{MC/MC}
      \end{array}
    \]
    Only the \textbf{bold} combinations survive.
  \end{block}
  \note{Because the two channels share the same exit rate beta but couple only
    at the entrances, we can think of each channel as a standard single-lane
    TASEP with its own effective entrance rate. Channel 1 feels alpha_1, channel
    2 feels alpha_2. Each channel independently sits in LD, HD or MC, giving
    nine candidate combinations. The job of the mean-field calculation is to
    check each of the nine for self-consistency and see which ones actually
    exist. We will find that only LD/LD and HD/LD survive.}
\end{frame}

\begin{frame}{Symmetric phases: set-up}
  \small
  Notation: $p_i$ = density at site $i$ of channel 1, $m_i$ = density at site
  $i$ of channel 2, both counted from their own entrance ($i=1,\dots,L$).
  So $m_1$ is channel 2's exit density, $p_L$ is channel 1's exit density.
  \begin{block}{Symmetric ansatz (eq 6-7)}
    \[
      J_1 = J_2, \quad \rho_1 = \rho_2, \quad \alpha_1 = \alpha_2, \qquad
      p_i = m_{L-i+1}.
    \]
  \end{block}
  \begin{itemize}
    \item Both channels in the same single-lane phase.
    \item Symmetry + the exit current relation determine the effective rates
      \emph{self-consistently}.
    \item Key relation (exit current of channel 2, eq 8/11):
      \[
        J_2 = \beta\, m_1 .
      \]
  \end{itemize}
  \note{Let me fix the notation, since this comes up everywhere. Channel 1 has
    densities p_i and channel 2 has densities m_i, both counted from their own
    entrance. So the coupling at the left end ties the entrance of channel 1,
    p_1, to the exit of channel 2, m_1; and at the right end ties channel 2's
    entrance m_L to channel 1's exit p_L. The exit current of channel 2 is
    beta times m_1, its exit-site density. This J_2 = beta m_1 relation is the
    workhorse of the whole calculation.}
\end{frame}

\begin{frame}{Symmetric MC phase (eqs 8--10)}
  \scriptsize
  \pause
  \textbf{Step 1: current fixes $m_1$ (eq 8).}
  In MC, $J_1 = J_2 = 1/4$; the exit current of channel 2 gives
  \[
    J_2 = \beta m_1 = \frac14 \quad\Rightarrow\quad m_1 = \frac{1}{4\beta}.
  \]
  \pause
  (This assumes $\beta \neq 0$; the physical range is $\beta>0$.)
  \pause
  \textbf{Step 2: effective entrance rate (eq 9).}
  \[
    \alpha_1 = \alpha(1 - m_1) = \alpha\left(1 - \frac{1}{4\beta}\right).
  \]
  \pause
  \textbf{Step 3: existence condition (eq 10).}
  MC needs $\alpha_1 > 1/2$ and $\beta > 1/2$:
  \[
    \alpha\left(1 - \frac{1}{4\beta}\right) > \frac12
    \quad\Longrightarrow\quad
    \alpha > \frac{2\beta}{4\beta - 1}
  \]
\end{frame}

\begin{frame}{Symmetric LD phase (eqs 11--13)}
  \scriptsize
  \pause
  \textbf{Step 1: current fixes $m_1$ (eq 11).}
  In LD, $J_1 = J_2 = \alpha_1(1-\alpha_1)$; exit current of channel 2:
  \[
    J_2 = \beta m_1 = \alpha_1(1-\alpha_1)
    \quad\Rightarrow\quad
    m_1 = \frac{\alpha_1(1-\alpha_1)}{\beta}.
  \]
  \pause
  \textbf{Step 2: self-consistency for $\alpha_1$ (eq 2).}
  \[
    \alpha_1 = \alpha(1 - m_1)
    = \alpha\left(1 - \frac{\alpha_1(1-\alpha_1)}{\beta}\right).
  \]
  A quadratic in $\alpha_1$; physical root (eq 12):
  \[
    \alpha_1 = \frac{\alpha + \beta - \sqrt{(\alpha+\beta)^2 - 4\alpha^2\beta}}{2\alpha}.
  \]
  \pause
  \textbf{Step 3: existence condition (eq 13).}
  LD needs $\alpha_1 < 1/2$ and $\alpha_1 < \beta$:
  \[
    \alpha < \frac{2\beta}{4\beta - 1}
  \]
\end{frame}

\begin{frame}{Symmetric HD phase: does not exist}
  \pause
  \begin{itemize}
    \item In HD, $J_1 = J_2 = \beta(1-\beta)$. From $J_2 = \beta m_1$:
      \[
        m_1 = 1 - \beta.
      \]
    \pause
    \item Effective entrance rate:
      \[
        \alpha_1 = \alpha(1 - m_1) = \alpha\beta.
      \]
    \pause
    \item HD requires $\alpha_1 > \beta$, i.e.\ $\alpha\beta > \beta$, i.e.\
      $\alpha > 1$. But $\alpha \le 1$ by definition.
  \end{itemize}
  \pause
  \begin{block}{Conclusion}
    \textbf{Symmetric HD phase is impossible} in the two-channel model with
    narrow entrances. Only MC and LD survive as symmetric phases.
  \end{block}
\end{frame}

\begin{frame}{Asymmetric phases: MC/HD and MC/LD don't exist}
  \small
  \pause
  \begin{itemize}
    \item \textbf{MC/HD} (eq 15): requires simultaneously $\beta>1/2$
      (for MC) and $\beta<1/2$ (for HD) --- contradictory.
    \pause
    \item \textbf{MC/LD} (eqs 16--19): MC needs $\alpha_1 > 1/2$, LD needs
      $\alpha_2 < 1/2$. Using $J_1 = \beta p_L = 1/4$ and
      $J_2 = \beta m_1 = \alpha_2(1-\alpha_2)$ leads to a condition (eq 19)
      that cannot hold for $0 < \alpha \le 1$, $0 < \beta \le 1$.
  \end{itemize}
  \pause
  \begin{block}{Conclusion}
    An asymmetric phase must pair \textbf{one channel in LD} with the other in
    a different phase: the candidate is HD/LD.
  \end{block}
\end{frame}

\begin{frame}{Asymmetric HD/LD phase (eqs 20--23)}
  \scriptsize
  \pause
  \textbf{Conditions (eq 20).}
  Channel 1 in HD: $\alpha_1 > \beta$, $\beta < 1/2$;\quad
  Channel 2 in LD: $\alpha_2 < \beta$, $\alpha_2 < 1/2$.
  \pause
  \textbf{Step 1: currents (eq 21).}
  \[
    J_1 = \beta p_L = \beta(1-\beta)
    \quad\Rightarrow\quad p_L = 1 - \beta,
    \qquad
    J_2 = \beta m_1 = \alpha_2(1-\alpha_2).
  \]
  \pause
  \textbf{Step 2: effective rates (eq 22).}
  \[
    \alpha_2 = \alpha(1 - p_L) = \alpha\beta, \qquad
    \alpha_1 = \alpha(1 - m_1) = \alpha(1 - \alpha + \alpha^2\beta).
  \]
  \pause
  \textbf{Step 3: existence (eq 23).}
  HD condition $\alpha_1 > \beta$:
  \[
    \alpha(1 - \alpha + \alpha^2\beta) > \beta
    \quad\Longrightarrow\quad
    \beta < \frac{\alpha}{1 + \alpha + \alpha^2}
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: set-up (eqs 24--26)}
  \small
  \pause
  Both channels in LD, but with \emph{different} densities and currents.
  \begin{block}{Conditions (eq 24)}
    $\alpha_1 < \beta$, $\alpha_1 < 1/2$ and $\alpha_2 < \beta$, $\alpha_2 < 1/2$.
  \end{block}
  \pause
  \begin{block}{Currents and effective rates (eqs 25--26)}
    \[
      J_1 = \beta p_L = \alpha_1(1-\alpha_1), \qquad
      J_2 = \beta m_1 = \alpha_2(1-\alpha_2),
    \]
    \[
      p_L = \frac{\alpha_1(1-\alpha_1)}{\beta}, \qquad
      m_1 = \frac{\alpha_2(1-\alpha_2)}{\beta},
    \]
    \[
      \alpha_1 = \alpha\left(1 - \frac{\alpha_2(1-\alpha_2)}{\beta}\right), \qquad
      \alpha_2 = \alpha\left(1 - \frac{\alpha_1(1-\alpha_1)}{\beta}\right).
    \]
  \end{block}
  \pause
  Two coupled quadratics in $\alpha_1, \alpha_2$. Simplify with symmetric and
  antisymmetric combinations (eq 27):
  \[
    S = \alpha_1 + \alpha_2, \qquad D = \alpha_1 - \alpha_2 \neq 0.
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: solve (eqs 27--31)}
  \scriptsize
  \pause
  \textbf{Step 1: difference of the two equations (eq 28).}
  \[
    D = \frac{\alpha}{\beta}\left(\alpha_1 - \alpha_1^2 - \alpha_2 + \alpha_2^2\right)
    = \frac{\alpha}{\beta} D (1 - S).
  \]
  Since $D \neq 0$ (asymmetric phase), we get
  \[
    S = 1 - \frac{\beta}{\alpha} \qquad \text{(eq 29)}
  \]
  \pause
  \textbf{Step 2: sum of the two equations (eq 30).}
  \[
    S = 2\alpha - \frac{\alpha}{\beta}\left[S - \frac{S^2 + D^2}{2}\right]
    \qquad\text{(using $\alpha_1^2 + \alpha_2^2 = (S^2+D^2)/2$).}
  \]
  \pause
  \textbf{Step 3: substitute $S = 1 - \beta/\alpha$ (eq 31).}
  \[
    D^2 = 1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2
  \]
\end{frame}

\begin{frame}{Asymmetric LD/LD phase: effective rates (eq 32)}
  \scriptsize
  From $S$ and $D$ we reconstruct the two effective entrance rates:
  \[
    \alpha_1 = \frac{S + D}{2}, \qquad \alpha_2 = \frac{S - D}{2}.
  \]
  \pause
  \begin{align*}
    \alpha_1 &= \frac{1}{2}\left[1 - \frac{\beta}{\alpha} + \sqrt{1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2}\right], \\[0.4em]
    \alpha_2 &= \frac{1}{2}\left[1 - \frac{\beta}{\alpha} - \sqrt{1 - 4\beta + 2\frac{\beta}{\alpha} - 3\left(\frac{\beta}{\alpha}\right)^2}\right].
  \end{align*}
  \pause
  \begin{block}{Existence (eq 33)}
    Real roots need $D^2 > 0$; together with the LD conditions (eq 24):
    \[
      \frac{\alpha}{1 + \alpha + \alpha^2} < \beta <
        \frac{\alpha(1-2\alpha) + 2\alpha\sqrt{\alpha^2 - \alpha + 1}}{3}
    \]
  \end{block}
\end{frame}

\begin{frame}{Mean-field phase diagram (summary)}
  \begin{columns}
    \column{0.5\textwidth}
      \centering
      \includegraphics[width=0.95\textwidth]{fig2/phase_diagram_full.png}
    \column{0.5\textwidth}
      \small
      \begin{tabular}{ll}
        \toprule
        Phase & MFT boundary \\
        \midrule
        MC & $\alpha > \dfrac{2\beta}{4\beta-1}$ (eq 10) \\
        LD & $\alpha < \dfrac{2\beta}{4\beta-1}$ (eq 13) \\
        HD & never (needs $\alpha>1$) \\
        HD/LD & $\beta < \dfrac{\alpha}{1+\alpha+\alpha^2}$ (eq 23) \\
        LD/LD & eq 33 (narrow band) \\
        \bottomrule
      \end{tabular}

      \vspace{0.5em}
      Transitions: LD$\leftrightarrow$MC and LD$\leftrightarrow$LD/LD are
      \emph{continuous}; HD/LD$\leftrightarrow$LD/LD is \emph{first order}.
  \end{columns}
\end{frame}

\begin{frame}{Take stock: what mean-field predicts}
  \begin{enumerate}
    \item Two symmetric phases (LD, MC) --- HD impossible.
    \item Two symmetry-broken phases: HD/LD (first order) and LD/LD
      (continuous), in a \emph{very narrow} band of $\beta$.
    \item The phase diagram is fully determined by self-consistency of the
      effective entrance rates $\alpha_1, \alpha_2$.
  \end{enumerate}
  \pause
  \begin{block}{The message to verify}
    Mean-field predicts \emph{where} transitions happen. Are these positions
    exact? For single-lane TASEP they are. Here, the boundary coupling may
    introduce correlations mean-field misses.
  \end{block}
\end{frame}

\begin{frame}{Why does mean-field fail here?}
  \begin{block}{The source of the discrepancy}
    Mean-field assumes \emph{no correlations} between sites (eq 1). The
    narrow-entrance coupling ties each entrance to the other channel's exit
    site --- a strongly-correlated, local ``impurity'' at the boundary that
    mean-field cannot capture.
  \end{block}
  \begin{itemize}
    \item These boundary correlations \emph{propagate into the bulk} and shift
      the effective phase boundaries.
    \item The paper's own conclusion (Sec.~4): symmetry breaking in this model
      is a \emph{boundary-induced} phenomenon --- an effective impurity at the
      boundary.
    \item Whether the MFT-vs-MC deviation persists in the thermodynamic limit
      is still open (needs longer runs at $L=800$).
  \end{itemize}
\end{frame}

\section{Spontaneous symmetry breaking}
\begin{frame}{SSB: the robust order parameter}
  \small
  \begin{block}{Why a robust order parameter is needed}
    Time-averaged $|\rho_1 - \rho_2|$ is \emph{washed out} because the broken
    states flip between $(+, -)$ and $(-, +)$ on long time scales. Robust
    choice: $\mathrm{std}(\rho_1 - \rho_2)$ over an ensemble of replicas.
  \end{block}
  \begin{columns}
    \column{0.40\textwidth}
      \centering
      \includegraphics[width=\textwidth]{ssb_order_vs_beta.png}
    \column{0.60\textwidth}
      \pause
      \begin{itemize}
        \item Long-run scan at $L=1000$, $\alpha=0.9$: order parameter
          decreases monotonically with $\beta$ ($0.09 \to 0.02$ from
          $\beta{=}0.05$ to $0.30$).
        \item SSB confined to \emph{low} $\beta$ in our MC (vs the paper's
          $\beta\approx0.23$ at the same $L$).
      \end{itemize}
  \end{columns}
\end{frame}

\section{Computational methods}
\begin{frame}{Simulations: BKL and the GPU ensemble}
  \begin{columns}
    \column{0.5\textwidth}
      \textbf{BKL (Bortz--Kalos--Lebowitz)}
      \begin{itemize}
        \item Rejection-free: pick events proportional to their rates.
        \item Classic double-loop selection: cost $\propto n_{\mathrm{active}}$.
        \item Fenwick-tree selection in $O(\log L)$: \textbf{$\sim$4$\times$
          faster, identical physics}.
      \end{itemize}
      \vspace{0.5em}
      \begin{tabular}{lc}
        \toprule
        Kernel & Msteps/s \\
        \midrule
        classic BKL & 1.2 \\
        Fenwick BKL & 4.8 \\
        \bottomrule
      \end{tabular}
    \column{0.5\textwidth}
      \textbf{GPU: one thread per replica}
      \[
        \text{GPU}:\; \mathrm{rep}_1 \oplus \mathrm{rep}_2 \oplus \cdots \oplus \mathrm{rep}_N
      \]
      \begin{itemize}
        \item A serial chain maps poorly to the GPU; an \emph{ensemble} maps
          perfectly.
        \item Thousands of replicas per $\beta$ --- exactly what $P(\rho_1,\rho_2)$
          needs.
      \end{itemize}
      \vspace{0.5em}
      \begin{tabular}{lc}
        \toprule
        Backend & Msteps/s \\
        \midrule
        CPU, 1 thread & 4.4 \\
        CPU, 12 cores & 29.6 \\
        GPU ensemble (2060 SUPER) & $\sim$85 \\
        \bottomrule
      \end{tabular}
  \end{columns}
\end{frame}

\begin{frame}{One run $\to$ all figures}
  \begin{itemize}
    \item \texttt{run\_all\_gpu.py}: one GPU ensemble scan produces ALL MC data
      (phase diagram, currents, densities, $P(\rho_1,\rho_2)$, SSB) saved to
      \texttt{results/gpu/*.npz}.
    \item \texttt{plot\_all.py}: plots every figure by \emph{reading} the saved
      data --- no MC at plot time.
    \item Reproducible: fixed seeds, checkpointed chunks (crash-safe).
  \end{itemize}
  \pause
  \begin{block}{Cost model (BKL vs Fenwick vs GPU)}
    Classic BKL $\propto n_{\mathrm{active}}$ (fast at low density), Fenwick/GPU
    $\propto \log L$ (wins at high density). The paths \emph{cross} in $\beta$.
  \end{block}
\end{frame}

\section{Conclusions \& Outlook}
\begin{frame}{Summary}
  \begin{itemize}
    \item Reproduced the two-channel ASEP with narrow entrances: model, MFT
      phase diagram, currents, densities, $P(\rho_1,\rho_2)$.
    \item Mean-field predicts two symmetric (LD, MC) and two broken (HD/LD,
      LD/LD) phases via self-consistent effective entrance rates.
    \item Confirmed the paper's message: MFT transition positions are
      \emph{not exact} --- the boundary coupling creates correlations MFT
      misses.
    \item SSB characterized robustly with $\mathrm{std}(\rho_1-\rho_2)$ and the
      ensemble $P(\rho_1,\rho_2)$; strong finite-size dependence.
    \item GPU ensemble ($\sim$85 Msteps/s) enables thousands of replicas and
      long runs.
  \end{itemize}
\end{frame}

\begin{frame}{Open questions}
  \begin{itemize}
    \item Does the LD/LD band survive at $L=1000$ and move with $L$? (Paper
      suggests it vanishes in the thermodynamic limit.)
    \item Does the MFT-vs-MC deviation shrink with $L$ or persist (genuine
      mean-field failure)? Needs longer runs at $L=800$.
    \item Can going beyond 2nd order / adding boundary-correlation parameters
      fix MFT?
    \item Extensions: asymmetric rates, wider entrances, 3 channels?
  \end{itemize}
  \vspace{0.8em}

\end{frame}
